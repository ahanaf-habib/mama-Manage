CREATE DATABASE IF NOT EXISTS `smart_apartment_management_system`   DEFAULT CHARACTER SET utf8mb4   COLLATE utf8mb4_general_ci;
USE `smart_apartment_management_system`;
-- ------------------------------------------------------------
-- Drop existing tables so this file can be imported cleanly

DROP TABLE IF EXISTS `move_out_requests`;
DROP TABLE IF EXISTS `apartment_requests`;
DROP TABLE IF EXISTS `payments`;
DROP TABLE IF EXISTS `bills`;
DROP TABLE IF EXISTS `maintenance_requests`;
DROP TABLE IF EXISTS `complaints`;
DROP TABLE IF EXISTS `notices`;
DROP TABLE IF EXISTS `tenancies`;
DROP TABLE IF EXISTS `apartments`;
DROP TABLE IF EXISTS `staff`;
DROP TABLE IF EXISTS `tenants`;
DROP TABLE IF EXISTS `owners`;
-- ------------------------------------------------------------
-- 1. OWNERS
-- ------------------------------------------------------------

CREATE TABLE `owners` (   `owner_id` int(11) NOT NULL AUTO_INCREMENT,   `owner_name` varchar(100) NOT NULL,   `phone` varchar(30) NOT NULL,   `email` varchar(150) NOT NULL,   `username` varchar(50) NOT NULL,   `password` varchar(255) NOT NULL,   PRIMARY KEY (`owner_id`),   UNIQUE KEY `email` (`email`),   UNIQUE KEY `username` (`username`) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 2. TENANTS
-- ------------------------------------------------------------

CREATE TABLE `tenants` (   `tenant_id` int(11) NOT NULL AUTO_INCREMENT,   `tenant_name` varchar(100) NOT NULL,   `phone` varchar(30) NOT NULL,   `email` varchar(150) NOT NULL,   `emergency_contact` varchar(100) DEFAULT NULL,   `id_reference` varchar(100) NOT NULL,   `username` varchar(50) NOT NULL,   `password` varchar(255) NOT NULL,   PRIMARY KEY (`tenant_id`),   UNIQUE KEY `email` (`email`),   UNIQUE KEY `id_reference` (`id_reference`),   UNIQUE KEY `username` (`username`) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 3. STAFF
-- ------------------------------------------------------------

CREATE TABLE `staff` (   `staff_id` int(11) NOT NULL AUTO_INCREMENT,   `staff_name` varchar(100) NOT NULL,   `phone` varchar(30) NOT NULL,   `experties` varchar(150) NOT NULL,   PRIMARY KEY (`staff_id`) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- 4. APARTMENTS
-- ------------------------------------------------------------

CREATE TABLE `apartments` (   `apartment_id` int(11) NOT NULL AUTO_INCREMENT,   `owner_id` int(11) NOT NULL,   `apartment_number` varchar(20) NOT NULL,   `floor_level` int(11) NOT NULL,   `description` text DEFAULT NULL,   `monthly_rent` decimal(10,2) NOT NULL,   `status` enum('Available','Occupied','Under Maintenance') NOT NULL DEFAULT 'Available',   PRIMARY KEY (`apartment_id`),   KEY `owner_id` (`owner_id`),   CONSTRAINT `apartments_ibfk_1`     FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------

-- ------------------------------------------------------------
-- 5. TENANCIES
-- ------------------------------------------------------------

CREATE TABLE `tenancies` (   `tenancy_id` int(11) NOT NULL AUTO_INCREMENT,   `tenant_id` int(11) NOT NULL,   `apartment_id` int(11) NOT NULL,   `move_in_date` date NOT NULL,   `move_out_date` date DEFAULT NULL,   `tenancy_status` enum('Active','Ended') NOT NULL DEFAULT 'Active',   PRIMARY KEY (`tenancy_id`),   KEY `tenant_id` (`tenant_id`),   KEY `apartment_id` (`apartment_id`),   CONSTRAINT `tenancies_ibfk_1`     FOREIGN KEY (`tenant_id`) REFERENCES `tenants` (`tenant_id`)     ON DELETE CASCADE,   CONSTRAINT `tenancies_ibfk_2`     FOREIGN KEY (`apartment_id`) REFERENCES `apartments` (`apartment_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 6. BILLS
-- ------------------------------------------------------------

CREATE TABLE `bills` (   `bill_id` int(11) NOT NULL AUTO_INCREMENT,   `tenancy_id` int(11) NOT NULL,   `billing_month` varchar(20) NOT NULL,   `issue_date` date NOT NULL,   `due_date` date NOT NULL,   `rent_amount` decimal(10,2) NOT NULL,   `utility_charge` decimal(10,2) NOT NULL DEFAULT 0.00,   `maintenance_service_charge` decimal(10,2) NOT NULL DEFAULT 0.00,   `other_charge` decimal(10,2) NOT NULL DEFAULT 0.00,   `total_amount` decimal(10,2) NOT NULL,   `bill_status` enum('Paid','Partially Paid','Unpaid','Overdue') NOT NULL DEFAULT 'Unpaid',   PRIMARY KEY (`bill_id`),   UNIQUE KEY `tenancy_month` (`tenancy_id`,`billing_month`),   KEY `tenancy_id` (`tenancy_id`),   CONSTRAINT `bills_ibfk_1`     FOREIGN KEY (`tenancy_id`) REFERENCES `tenancies` (`tenancy_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 7. PAYMENTS
-- ------------------------------------------------------------

CREATE TABLE `payments` (   `payment_id` int(11) NOT NULL AUTO_INCREMENT,   `bill_id` int(11) NOT NULL,   `payment_amount` decimal(10,2) NOT NULL,   `payment_date` date NOT NULL,   `payment_method` enum('Cash','bKash','Bank Transfer') NOT NULL,   `transaction_reference_id` varchar(100) DEFAULT NULL,   `payment_status` enum('Completed','Pending','Failed') NOT NULL DEFAULT 'Completed',   PRIMARY KEY (`payment_id`),   KEY `bill_id` (`bill_id`),   CONSTRAINT `payments_ibfk_1`     FOREIGN KEY (`bill_id`) REFERENCES `bills` (`bill_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 8. COMPLAINTS
-- ------------------------------------------------------------

CREATE TABLE `complaints` (   `complaint_id` int(11) NOT NULL AUTO_INCREMENT,   `tenant_id` int(11) NOT NULL,   `apartment_id` int(11) NOT NULL,   `category` varchar(100) NOT NULL,   `description` text NOT NULL,   `priority` enum('Low','Medium','High') NOT NULL DEFAULT 'Medium',   `submission_date` date NOT NULL,   `complaint_status` enum('Pending','Assigned','In Progress','Resolved') NOT NULL DEFAULT 'Pending',   PRIMARY KEY (`complaint_id`),   KEY `tenant_id` (`tenant_id`),   KEY `apartment_id` (`apartment_id`),   CONSTRAINT `complaints_ibfk_1`     FOREIGN KEY (`tenant_id`) REFERENCES `tenants` (`tenant_id`)     ON DELETE CASCADE,   CONSTRAINT `complaints_ibfk_2`     FOREIGN KEY (`apartment_id`) REFERENCES `apartments` (`apartment_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 9. MAINTENANCE_REQUESTS
-- ------------------------------------------------------------

CREATE TABLE `maintenance_requests` (   `maintenance_id` int(11) NOT NULL AUTO_INCREMENT,   `complaint_id` int(11) NOT NULL,   `staff_id` int(11) DEFAULT NULL,   `estimated_cost` decimal(10,2) NOT NULL DEFAULT 0.00,   `maintenance_status` enum('Assigned','In Progress','Completed') NOT NULL DEFAULT 'Assigned',   `completion_date` date DEFAULT NULL,   `maintenance_notes` text DEFAULT NULL,   PRIMARY KEY (`maintenance_id`),   KEY `complaint_id` (`complaint_id`),   KEY `staff_id` (`staff_id`),   CONSTRAINT `maintenance_requests_ibfk_1`     FOREIGN KEY (`complaint_id`) REFERENCES `complaints` (`complaint_id`)     ON DELETE CASCADE,   CONSTRAINT `maintenance_requests_ibfk_2`     FOREIGN KEY (`staff_id`) REFERENCES `staff` (`staff_id`)     ON DELETE SET NULL ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 10. NOTICES
-- ------------------------------------------------------------

CREATE TABLE `notices` (   `notice_id` int(11) NOT NULL AUTO_INCREMENT,   `owner_id` int(11) NOT NULL,   `message_content` text NOT NULL,   `published_at` timestamp NOT NULL DEFAULT current_timestamp(),   `target_type` enum('Everyone','Apartment','Floor') NOT NULL DEFAULT 'Everyone',   `apartment_id` int(11) DEFAULT NULL,   `floor_number` int(11) DEFAULT NULL,   PRIMARY KEY (`notice_id`),   KEY `owner_id` (`owner_id`),   KEY `apartment_id` (`apartment_id`),   CONSTRAINT `notices_ibfk_1`     FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`)     ON DELETE CASCADE,   CONSTRAINT `notices_ibfk_2`     FOREIGN KEY (`apartment_id`) REFERENCES `apartments` (`apartment_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- ------------------------------------------------------------
-- 11. APARTMENT_REQUESTS
-- ------------------------------------------------------------

CREATE TABLE `apartment_requests` (   `request_id` int(11) NOT NULL AUTO_INCREMENT,   `tenant_id` int(11) NOT NULL,   `apartment_id` int(11) NOT NULL,   `owner_id` int(11) NOT NULL,   `requested_at` datetime NOT NULL DEFAULT current_timestamp(),   `responded_at` datetime DEFAULT NULL,   `status` enum('Pending','Approved','Rejected') NOT NULL DEFAULT 'Pending',   PRIMARY KEY (`request_id`),   KEY `tenant_id` (`tenant_id`),   KEY `apartment_id` (`apartment_id`),   KEY `owner_id` (`owner_id`),   CONSTRAINT `apartment_requests_ibfk_1`     FOREIGN KEY (`tenant_id`) REFERENCES `tenants` (`tenant_id`)     ON DELETE CASCADE,   CONSTRAINT `apartment_requests_ibfk_2`     FOREIGN KEY (`apartment_id`) REFERENCES `apartments` (`apartment_id`)     ON DELETE CASCADE,   CONSTRAINT `apartment_requests_ibfk_3`     FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
-- ------------------------------------------------------------
-- 12. MOVE_OUT_REQUESTS
-- ------------------------------------------------------------

CREATE TABLE `move_out_requests` (   `request_id` int(11) NOT NULL AUTO_INCREMENT,   `tenancy_id` int(11) NOT NULL,   `tenant_id` int(11) NOT NULL,   `owner_id` int(11) NOT NULL,   `requested_at` datetime NOT NULL DEFAULT current_timestamp(),   `proposed_move_out_date` date NOT NULL,   `responded_at` datetime DEFAULT NULL,   `status` enum('Pending','Approved','Rejected','Cancelled') NOT NULL DEFAULT 'Pending',   `initiated_by` enum('Tenant','Owner') NOT NULL DEFAULT 'Tenant',   PRIMARY KEY (`request_id`),   KEY `tenancy_id` (`tenancy_id`),   KEY `tenant_id` (`tenant_id`),   KEY `owner_id` (`owner_id`),   CONSTRAINT `move_out_requests_ibfk_1`     FOREIGN KEY (`tenancy_id`) REFERENCES `tenancies` (`tenancy_id`)     ON DELETE CASCADE,   CONSTRAINT `move_out_requests_ibfk_2`     FOREIGN KEY (`tenant_id`) REFERENCES `tenants` (`tenant_id`)     ON DELETE CASCADE,   CONSTRAINT `move_out_requests_ibfk_3`     FOREIGN KEY (`owner_id`) REFERENCES `owners` (`owner_id`)     ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
